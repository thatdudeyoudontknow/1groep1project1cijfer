! =====================================================================
! SW-BG2-01 - L2 switch 2 (CBS350-8T-E-2G) - Bijgebouw 2
! gi1 trunk <- SW-ELB-01 Gi1/0/4 · gi2 trunk <- SW-ELB-02 Gi1/0/4 · gi3 access 11
! Plak per blok in "configure terminal"
! =====================================================================

! ---------- 1. Basis ----------
hostname SW-BG2-01
username admin password Waarisjos! privilege 15

! ---------- 2. VLANs ----------
vlan database
vlan 10-13,999
exit
interface vlan 10
 name Netwerkbeheer
interface vlan 11
 name Medewerkers
interface vlan 12
 name Studenten
interface vlan 13
 name WiFi-BYOD
interface vlan 999
 name Ongebruikt
exit

! ---------- 3. Spanning tree (zelfde MST-regio als de 3650's) ----------
spanning-tree mode mst
spanning-tree mst configuration
 name HEH
 revision 1
 instance 1 vlan 10-13,99
exit

! ---------- 4. Poorten ----------
interface GigabitEthernet1
 description "Trunk naar SW-ELB-01 Gi1/0/4"
 switchport mode trunk
 switchport trunk allowed vlan none
 switchport trunk allowed vlan add 10-13
 no shutdown
interface GigabitEthernet2
 description "Trunk naar SW-ELB-02 Gi1/0/4"
 switchport mode trunk
 switchport trunk allowed vlan none
 switchport trunk allowed vlan add 10-13
 no shutdown
interface GigabitEthernet3
 description "Access VLAN 11 -> K1P2-4 -> tafel 6b4"
 switchport mode access
 switchport access vlan 11
 spanning-tree portfast
 no shutdown
interface range GigabitEthernet4-10
 description Ongebruikt
 switchport mode access
 switchport access vlan 999
 shutdown
exit

! ---------- 5. Beheeradres ----------
interface vlan 1
 no ip address dhcp
interface vlan 10
 ip address 10.0.0.5 255.255.255.0
exit
ip default-gateway 10.0.0.1

! ---------- 6. Beheer (SSH) ----------
ip ssh server
no ip telnet server
end
write memory
