! =====================================================================
! R-ELB-02 - Router 2 (ISR 4321 + NIM-ES2-4) - edge-router Elburg
! Alle links op de NIM-ES2-4 (switchpoorten) -> IP staat op de SVI
!   Gi0/1/0  VLAN 111  -> FW-ELB-01 lan2      10.1.3.10/29    (FW 10.1.3.9)
!   Gi0/1/1  VLAN 112  -> SW-ELB-01 Gi1/0/2   192.168.0.5/30  (SW 192.168.0.6)
!   Gi0/1/2  VLAN 121  -> FW-ELB-02 lan2      10.1.3.26/29    (FW 10.1.3.25)
!   Gi0/1/3  VLAN 122  -> SW-ELB-02 Gi1/0/2   192.168.0.13/30 (SW 192.168.0.14)
!   Console  -> TC ge2
! Plak per blok in "configure terminal"
! =====================================================================

! ---------- 0. (Optioneel) Volledig wissen vóór herinstallatie ----------
! In privileged EXEC (niet in configure terminal):
!   write erase
!   delete /force flash:vlan.dat      (alleen als dit bestand bestaat)
!   reload                            (bij "save?" -> no)
! Sla daarna de initial configuration dialog over (no).

! ---------- 1. Basis ----------
hostname R-ELB-02
no ip domain-lookup
ip domain-name heh.local
enable secret <WACHTWOORD>
username admin privilege 15 secret <WACHTWOORD>
service password-encryption

! ---------- 2. Ingebouwde poorten: niet in gebruik ----------
interface GigabitEthernet0/0/0
 description Ongebruikt (alle links zitten op de NIM)
 shutdown
interface GigabitEthernet0/0/1
 description Ongebruikt (alle links zitten op de NIM)
 shutdown
exit

! ---------- 3. VLANs (lokaal op de NIM, één VLAN per link) ----------
vlan 111
 name FW1-transit
vlan 112
 name SW1-transit
vlan 121
 name FW2-transit
vlan 122
 name SW2-transit
exit

! ---------- 4. NIM-poorten ----------
interface GigabitEthernet0/1/0
 description ge0/1/0 - FW-ELB-01 lan2
 switchport mode access
 switchport access vlan 111
 spanning-tree portfast
 no shutdown
interface GigabitEthernet0/1/1
 description ge0/1/1 - SW-ELB-01 Gi1/0/2
 switchport mode access
 switchport access vlan 112
 spanning-tree portfast
 no shutdown
interface GigabitEthernet0/1/2
 description ge0/1/2 - FW-ELB-02 lan2
 switchport mode access
 switchport access vlan 121
 spanning-tree portfast
 no shutdown
interface GigabitEthernet0/1/3
 description ge0/1/3 - SW-ELB-02 Gi1/0/2
 switchport mode access
 switchport access vlan 122
 spanning-tree portfast
 no shutdown
exit

! ---------- 5. SVI's ----------
interface Vlan1
 shutdown
interface Vlan111
 description Transit naar FW-ELB-01
 ip address 10.1.3.10 255.255.255.248
 no shutdown
interface Vlan112
 description Transit naar SW-ELB-01
 ip address 192.168.0.5 255.255.255.252
 no shutdown
interface Vlan121
 description Transit naar FW-ELB-02
 ip address 10.1.3.26 255.255.255.248
 no shutdown
interface Vlan122
 description Transit naar SW-ELB-02
 ip address 192.168.0.13 255.255.255.252
 no shutdown
exit

! ---------- 6. Statische routes (exit-interface + next-hop) ----------
! Internet: primair FW1, backup FW2
ip route 0.0.0.0 0.0.0.0 Vlan111 10.1.3.9
ip route 0.0.0.0 0.0.0.0 Vlan121 10.1.3.25 10
! Datacenter (achter de firewalls)
ip route 10.1.1.0 255.255.255.0 Vlan111 10.1.3.9
ip route 10.1.1.0 255.255.255.0 Vlan121 10.1.3.25 10
! Rest van de DMZ (voorkomt een lus als er andere hosts in 10.1.3.0/24 staan)
ip route 10.1.3.0 255.255.255.0 Vlan111 10.1.3.9
ip route 10.1.3.0 255.255.255.0 Vlan121 10.1.3.25 10
! Interne netwerken: primair SW-ELB-01, backup SW-ELB-02
ip route 10.0.0.0 255.0.0.0 Vlan112 192.168.0.6
ip route 10.0.0.0 255.0.0.0 Vlan122 192.168.0.14 10
ip route 192.168.0.0 255.255.255.0 Vlan112 192.168.0.6
ip route 192.168.0.0 255.255.255.0 Vlan122 192.168.0.14 10

! ---------- Beheer (SSH) ----------
crypto key generate rsa modulus 2048
ip ssh version 2
line con 0
 exec-timeout 10 0
 logging synchronous
 login local
line vty 0 4
 exec-timeout 10 0
 login local
 transport input ssh
end
write memory
